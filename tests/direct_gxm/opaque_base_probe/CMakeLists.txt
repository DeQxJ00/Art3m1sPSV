cmake_minimum_required(VERSION 3.16)
project(opaque_base_probe LANGUAGES C CXX)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(opaque_base_probe ../opaque_base_probe.cpp ${ROOT}/host-direct/src/gpu.cpp)
target_include_directories(opaque_base_probe PRIVATE ${ROOT}/host-direct/src ${ROOT}/host)
target_compile_definitions(opaque_base_probe PRIVATE DIRECT_DEFERRED_FINISH_PROBE=1)
target_link_options(opaque_base_probe PRIVATE "-Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/../video_yuva_probe/sce_gap.ld")
target_link_libraries(opaque_base_probe PRIVATE SceGxm_stub SceDisplay_stub SceCtrl_stub -Wl,--whole-archive pthread -Wl,--no-whole-archive m c)
vita_create_self(opaque_base_probe.self opaque_base_probe UNSAFE)
vita_create_vpk(opaque_base_probe.vpk ART3OPB01 opaque_base_probe.self VERSION "01.00" NAME "GXM opaque base probe")
