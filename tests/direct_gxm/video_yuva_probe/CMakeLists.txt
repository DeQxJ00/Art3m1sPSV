cmake_minimum_required(VERSION 3.16)
project(video_yuva_probe LANGUAGES C CXX)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(video_yuva_probe ../video_yuva_probe.cpp ${ROOT}/host-direct/src/gpu.cpp)
target_include_directories(video_yuva_probe PRIVATE ${ROOT}/host-direct/src ${ROOT}/host)
target_compile_definitions(video_yuva_probe PRIVATE DIRECT_DEFERRED_FINISH_PROBE=1)
target_link_libraries(video_yuva_probe PRIVATE SceGxm_stub SceDisplay_stub SceCtrl_stub -Wl,--whole-archive pthread -Wl,--no-whole-archive m c)
vita_create_self(video_yuva_probe.self video_yuva_probe UNSAFE)
vita_create_vpk(video_yuva_probe.vpk ART3YUV01 video_yuva_probe.self VERSION "01.00" NAME "GXM YUVA probe")
