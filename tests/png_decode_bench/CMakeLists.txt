cmake_minimum_required(VERSION 3.16)
project(png_decode_bench C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(Z "${ROOT}/build/png-bench-zlib")
set(P "${ROOT}/build/png-bench-libpng")
configure_file("${P}/scripts/pnglibconf.h.prebuilt" "${CMAKE_CURRENT_BINARY_DIR}/pnglibconf.h" COPYONLY)
add_compile_options(-O3 -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard -ffunction-sections -fdata-sections)
set(zsources adler32.c compress.c crc32.c deflate.c infback.c inffast.c inflate.c inftrees.c trees.c uncompr.c zutil.c)
list(TRANSFORM zsources PREPEND "${Z}/")
add_library(bench_z STATIC ${zsources})
target_include_directories(bench_z PRIVATE "${Z}")
set(psources png.c pngerror.c pngget.c pngmem.c pngpread.c pngread.c pngrio.c pngrtran.c pngrutil.c pngset.c pngtrans.c pngwio.c pngwrite.c pngwtran.c pngwutil.c arm/arm_init.c arm/filter_neon_intrinsics.c arm/palette_neon_intrinsics.c)
list(TRANSFORM psources PREPEND "${P}/")
add_library(bench_png STATIC ${psources})
target_compile_definitions(bench_png PRIVATE PNG_ARM_NEON_OPT=2)
target_include_directories(bench_png PRIVATE "${P}" "${Z}" "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(png_bench bench.c)
target_include_directories(png_bench PRIVATE "${P}" "${Z}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(png_bench PRIVATE -include zlib.h)
target_link_options(png_bench PRIVATE -Wl,--gc-sections)
target_link_libraries(png_bench bench_png bench_z "${ROOT}/build/png-bench-rust/armv7-sony-vita-newlibeabihf/release/libpng_decode_bench.a" ScePower_stub SceDisplay_stub -Wl,--whole-archive pthread -Wl,--no-whole-archive m c)
vita_create_self(png_bench.self png_bench UNSAFE)
